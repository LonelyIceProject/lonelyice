cmake_minimum_required(VERSION 3.22)
project(StorageCheckTests LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 20)
set(launcher "${CMAKE_CURRENT_SOURCE_DIR}/../../src")
set(LONELYICE_CORE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../external/azerothcore" CACHE PATH "Core tree providing fmt headers")
add_executable(storage-check-test test.cpp "${launcher}/launcher/StorageCheck.cpp" "${launcher}/common/Platform.cpp")
target_include_directories(storage-check-test PRIVATE "${launcher}/common" "${launcher}/launcher" "${LONELYICE_CORE_DIR}/deps/fmt/include")
target_compile_definitions(storage-check-test PRIVATE FMT_HEADER_ONLY)
if(WIN32)
  target_link_libraries(storage-check-test PRIVATE shell32 ole32 uuid psapi)
endif()
add_test(NAME storage-check COMMAND storage-check-test)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/tmp")
set_tests_properties(storage-check PROPERTIES ENVIRONMENT
  "TEMP=${CMAKE_CURRENT_BINARY_DIR}/tmp;TMP=${CMAKE_CURRENT_BINARY_DIR}/tmp;TMPDIR=${CMAKE_CURRENT_BINARY_DIR}/tmp")
