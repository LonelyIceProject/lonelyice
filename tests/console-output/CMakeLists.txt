cmake_minimum_required(VERSION 3.22)
project(ConsoleOutputTests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
set(common "${CMAKE_CURRENT_LIST_DIR}/../../src/common")
add_executable(console-output-test test.cpp "${common}/Platform.cpp")
target_compile_features(console-output-test PRIVATE cxx_std_20)
target_include_directories(console-output-test PRIVATE "${common}")
target_link_libraries(console-output-test PRIVATE Threads::Threads)
if(WIN32)
  target_link_libraries(console-output-test PRIVATE shell32 ole32 uuid psapi)
endif()
if(MSVC)
  target_compile_options(console-output-test PRIVATE /utf-8)
endif()
add_test(NAME console-output COMMAND console-output-test)
